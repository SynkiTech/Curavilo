   198→   198→        ...parsed,
   199→   199→        nodes: parsed.nodes?.map((n) => ({ id: n.id, type: n.type as BuilderNode["type"], config: Object.fromEntries(Object.entries(n.config ?? {}).map(([k, v]) => [k, String(v)])) })),
   200→   200→      },
   201→   201→    };
   202→   202→  } catch {
   203→   203→    return undefined;

   566→   566→function FacilityCredentialCard() {
   567→   567→  const [email, setEmail] = useState("");
   568→   568→  const [password, setPassword] = useState("");
   569→   569→  const [confirm, setConfirm] = useState("");
   570→   570→  const [busy, setBusy] = useState(false);
   571→   571→  const [loading, setLoading] = useState(true);
   572→   572→
   573→   573→  useEffect(() => {
   574→   574→    let active = true;
   575→   575→    apiGet<{ credentialEmail: string | null }>("/api/facility-credential")
   576→   576→      .then((r) => { if (active) { setEmail(r.credentialEmail ?? ""); setLoading(false); } })
   577→   577→      .catch(() => active && setLoading(false));
   578→   578→    return () => { active = false; };
   579→   579→  }, []);
   580→   580→
   581→   581→  const save = async () => {
   582→   582→    if (password && password !== confirm) {
   583→   583→      toast.error("New passwords do not match.");
   584→   584→      return;
   585→   585→    }
   586→   586→    setBusy(true);
   587→   587→    try {
   588→   588→      await apiSend("PATCH", "/api/facility-credential", {
   589→   589→        ...(email ? { email } : {}),
   590→   590→        ...(password ? { password } : {}),
   591→   591→      });
   592→   592→      toast.success("Facility access credential updated.");
   593→   593→      setPassword("");
   594→   594→      setConfirm("");
   595→   595→    } catch (e) {
   596→   596→      toast.error((e as Error).message);
   597→   597→    } finally {
   598→   598→      setBusy(false);
   599→   599→    }
   600→   600→  };
   601→   601→
   602→   602→  return (
   603→   603→    <Card>
   604→   604→      <CardHeader><CardTitle className="text-sm">Facility access credential</CardTitle></CardHeader>
   605→   605→      <CardContent className="space-y-3">
   606→   606→        <p className="text-xs text-muted-foreground">
   607→   607→          This is the workspace credential used at the facility access gate (step 1 of sign-in). It opens
   608→   608→          the facility only - it never grants staff permissions. Owner-only, changeable at any time.
   609→   609→        </p>
   610→   610→        {loading ? <LoadingState /> : (
   611→   611→          <>
   612→   612→            <div className="space-y-1">
   613→   613→              <Label htmlFor="cred-email">Facility gate email</Label>

   618→   618→                <Label htmlFor="cred-pass">New gate password</Label>

   620→   620→              </div>
   621→   621→              <div className="space-y-1">
   622→   622→                <Label htmlFor="cred-confirm">Confirm password</Label>
   623→   623→                <Input id="cred-confirm" type="password" value={confirm} onChange={(e) => setConfirm(e.target.value)} />
   624→   624→              </div>
   625→   625→            </div>
   626→   626→            <Button className="w-full bg-teal-700 hover:bg-teal-800" disabled={busy || (!password && !email)} onClick={() => void save()}>
   627→   627→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null} Update facility credential
   628→   628→            </Button>
   629→   629→          </>
   630→   630→        )}

   632→   632→    </Card>
