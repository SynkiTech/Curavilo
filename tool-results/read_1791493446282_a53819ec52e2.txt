   114→const STORAGE_KEY = "curavilo.print.settings.v2";
   115→const MM_PER_IN = 25.4;
   116→const PX_PER_MM = 96 / 25.4;
   117→
   118→function loadSettings(): PrintSettings {
   119→  try {
   120→    const raw = window.localStorage.getItem(STORAGE_KEY);
   121→    if (!raw) return DEFAULTS;
   122→    const parsed = JSON.parse(raw) as Partial<PrintSettings>;
   123→    return { ...DEFAULTS, ...parsed };
   124→  } catch {
   125→    return DEFAULTS;
   126→  }
   127→}
   128→
   129→const isThermal = (key: PrintProfileKey) => key === "THERMAL_58MM" || key === "THERMAL_80MM" || key === "THERMAL_112MM";
   130→
   131→/** Resolve the effective sheet geometry from settings. */
   132→function resolveSheet(s: PrintSettings): { widthMm: number; heightMm: number | null; mono: boolean } {
   133→  if (s.profileKey === "CUSTOM") {
   134→    const w = s.customUnit === "in" ? s.customWidthMm * MM_PER_IN : s.customWidthMm;
   135→    const h = s.customUnit === "in" ? s.customHeightMm * MM_PER_IN : s.customHeightMm;
   136→    const [ew, eh] = s.orientation === "landscape" ? [Math.max(w, h), Math.min(w, h)] : [w, h];
   137→    return { widthMm: ew, heightMm: eh, mono: false };
   138→  }
   139→  const p = PRINT_PROFILES[s.profileKey as Exclude<PrintProfileKey, "CUSTOM">];
   140→  if (p.heightMm === null) return { widthMm: p.widthMm, heightMm: null, mono: p.mono };
   141→  const [ew, eh] = s.orientation === "landscape" ? [p.heightMm, p.widthMm] : [p.widthMm, p.heightMm];
   142→  return { widthMm: ew, heightMm: eh, mono: p.mono };
   143→}
   144→
   145→const fmtMm = (mm: number) => (Math.round(mm * 10) / 10).toString();

   147→export function DocumentRenderer({
   148→  branding,
   149→  data,
   150→  defaultProfile = "THERMAL_80MM",
   151→  onPrinted,
   152→}: {
   153→  branding: Branding;
   154→  data: ReceiptDocData;
   155→  defaultProfile?: PrintProfileKey;
   156→  onPrinted?: () => void;
   157→}) {
   158→  const [settings, setSettings] = useState<PrintSettings>(() => {
   159→    if (typeof window === "undefined") return { ...DEFAULTS, profileKey: defaultProfile };
   160→    const saved = loadSettings();
   161→    return saved.profileKey ? saved : { ...saved, profileKey: defaultProfile };
   162→  });
   163→  const printedRef = useRef(false);
   164→
   165→  useEffect(() => {
   166→    try {
   167→      window.localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
   168→    } catch {
   169→      // storage unavailable (private mode): settings stay in memory
   170→    }
   171→  }, [settings]);
   172→
   173→  const sheet = useMemo(() => resolveSheet(settings), [settings]);
   174→  const thermal = sheet.heightMm === null;
   175→  const paddingMm = thermal ? Math.min(settings.marginMm, 4) : settings.marginMm;
   176→  const fontSize = (sheet.mono ? 10 : 12) * (settings.scalePct / 100);
   177→
   178→  const update = (patch: Partial<PrintSettings>) => setSettings((prev) => ({ ...prev, ...patch }));
   179→
   180→  // Inject @page rule for the effective sheet while mounted
   181→  useEffect(() => {
   182→    const size = sheet.heightMm === null ? `${fmtMm(sheet.widthMm)}mm auto` : `${fmtMm(sheet.widthMm)}mm ${fmtMm(sheet.heightMm)}mm`;
   183→    const style = document.createElement("style");
   184→    style.setAttribute("data-print-profile", "true");
   185→    style.textContent = `@media print { @page { size: ${size}; margin: 0; } }`;
   186→    document.head.appendChild(style);
   187→    return () => {
   188→      style.remove();
   189→    };
   190→  }, [sheet]);
   191→
   192→  const doPrint = () => {
   193→    printedRef.current = true;
   194→    const originalTitle = document.title;
   195→    document.title = `${data.number} - ${branding.facilityName}`;
   196→    const restore = () => {
   197→      document.title = originalTitle;
   198→      window.removeEventListener("afterprint", restore);
   199→    };
   200→    window.addEventListener("afterprint", restore);
   201→    window.print();
   202→    onPrinted?.();
   203→    // Safety restore in case afterprint does not fire (some browsers + previews)
   204→    window.setTimeout(() => {
   205→      if (document.title !== originalTitle) restore();
   206→    }, 4000);
   207→  };
   208→
   209→  return (
   210→    <div className="space-y-3">
   211→      {/* Controls + preview (never printed) */}
   212→      <div className="no-print grid gap-3 sm:grid-cols-[1fr_auto]">
   213→        <div className="space-y-2.5 rounded-xl border bg-stone-50/70 p-3">
   214→          <div className="flex flex-wrap items-end gap-2">
   215→            <div className="space-y-1">
   216→              <Label className="text-[11px] text-muted-foreground">Paper size</Label>
   217→              <Select
   218→                value={settings.profileKey}
   219→                onValueChange={(v) => {
   220→                  const key = v as PrintProfileKey;
   221→                  update({ profileKey: key, orientation: isThermal(key) ? "portrait" : settings.orientation });
   222→                }}
   223→              >
   224→                <SelectTrigger className="w-[190px]" aria-label="Paper size">
   225→                  <SelectValue />
   226→                </SelectTrigger>
   227→                <SelectContent>
   228→                  {Object.values(PRINT_PROFILES).map((p) => (
   229→                    <SelectItem key={p.key} value={p.key}>{p.label}</SelectItem>
   230→                  ))}
   231→                  <SelectItem value="CUSTOM">Custom size…</SelectItem>
   232→                </SelectContent>
   233→              </Select>
   234→            </div>
   235→
   236→            {settings.profileKey === "CUSTOM" ? (
   237→              <>
   238→                <div className="space-y-1">
   239→                  <Label htmlFor="pp-w" className="text-[11px] text-muted-foreground">Width</Label>
   240→                  <Input
   241→                    id="pp-w"
   242→                    type="number"
   243→                    min={10}
   244→                    max={2000}
   245→                    step={settings.customUnit === "mm" ? 1 : 0.05}
   246→                    value={settings.customUnit === "mm" ? settings.customWidthMm : Math.round((settings.customWidthMm / MM_PER_IN) * 100) / 100}
   247→                    onChange={(e) => update({ customWidthMm: Math.max(10, Number(e.target.value) || 0) * (settings.customUnit === "in" ? MM_PER_IN : 1) })}
   248→                    className="h-9 w-24"
   249→                  />
   250→                </div>
   251→                <div className="space-y-1">
   252→                  <Label htmlFor="pp-h" className="text-[11px] text-muted-foreground">Height</Label>
   253→                  <Input
   254→                    id="pp-h"
   255→                    type="number"
   256→                    min={10}
   257→                    max={2000}
   258→                    step={settings.customUnit === "mm" ? 1 : 0.05}
   259→                    value={settings.customUnit === "mm" ? settings.customHeightMm : Math.round((settings.customHeightMm / MM_PER_IN) * 100) / 100}
   260→                    onChange={(e) => update({ customHeightMm: Math.max(10, Number(e.target.value) || 0) * (settings.customUnit === "in" ? MM_PER_IN : 1) })}
   261→                    className="h-9 w-24"
   262→                  />
   263→                </div>
   264→                <div className="space-y-1">
   265→                  <Label className="text-[11px] text-muted-foreground">Unit</Label>
   266→                  <Select value={settings.customUnit} onValueChange={(v) => update({ customUnit: v as Unit })}>
   267→                    <SelectTrigger className="w-[74px]" aria-label="Unit"><SelectValue /></SelectTrigger>
   268→                    <SelectContent>
   269→                      <SelectItem value="mm">mm</SelectItem>
   270→                      <SelectItem value="in">inch</SelectItem>
   271→                    </SelectContent>
   272→                  </Select>
   273→                </div>
   274→              </>
   275→            ) : null}
   276→
   277→            {!thermal ? (
   278→              <div className="space-y-1">
   279→                <Label className="text-[11px] text-muted-foreground">Orientation</Label>
   280→                <div className="flex h-9 overflow-hidden rounded-md border">
   281→                  {(["portrait", "landscape"] as Orientation[]).map((o) => (
   282→                    <button
   283→                      key={o}
   284→                      type="button"
   285→                      onClick={() => update({ orientation: o })}
   286→                      className={`px-3 text-xs transition ${settings.orientation === o ? "bg-teal-600 font-medium text-white" : "bg-white text-stone-500 hover:bg-teal-50"}`}
   287→                    >
   288→                      {o === "portrait" ? "Portrait" : "Landscape"}
   289→                    </button>
   290→                  ))}
   291→                </div>
   292→              </div>
   293→            ) : null}
   294→          </div>
   295→
   296→          <div className="flex flex-wrap items-end gap-2">
   297→            <div className="space-y-1">
   298→              <Label className="text-[11px] text-muted-foreground">Margin</Label>
   299→              <Select value={String(settings.marginMm)} onValueChange={(v) => update({ marginMm: Number(v) })}>
   300→                <SelectTrigger className="w-[150px]" aria-label="Margin"><SelectValue /></SelectTrigger>
   301→                <SelectContent>
   302→                  {MARGIN_PRESETS.map((m) => (
   303→                    <SelectItem key={m.mm} value={String(m.mm)}>{m.label}</SelectItem>
   304→                  ))}
   305→                </SelectContent>
   306→              </Select>
   307→            </div>
   308→            <div className="min-w-[170px] flex-1 space-y-1">
   309→              <Label htmlFor="pp-scale" className="text-[11px] text-muted-foreground">Text scale: {settings.scalePct}%</Label>
   310→              <input
   311→                id="pp-scale"
   312→                type="range"
   313→                min={80}
   314→                max={140}
   315→                step={5}
   316→                value={settings.scalePct}
   317→                onChange={(e) => update({ scalePct: Number(e.target.value) })}
   318→                className="h-9 w-full accent-teal-600"
   319→                aria-label="Text scale"
   320→              />
   321→            </div>
   322→            <Button variant="ghost" size="sm" onClick={() => setSettings({ ...DEFAULTS, profileKey: defaultProfile })} title="Reset print settings">
   323→              <RotateCcw className="mr-1.5 h-3.5 w-3.5" /> Reset
   324→            </Button>
   325→            <Button onClick={doPrint} size="sm" className="ml-auto">
   326→              <Printer className="mr-2 h-4 w-4" /> Print / Save PDF
   327→            </Button>
   328→          </div>
   329→
   330→          <p className="text-[11px] text-muted-foreground">
   331→            Sheet: {fmtMm(sheet.widthMm)} x {sheet.heightMm === null ? "auto" : fmtMm(sheet.heightMm)} mm · margin {fmtMm(paddingMm)} mm · your choice is remembered for next time.
   332→            In the browser dialog, pick the same paper size and set margins to None for a 1:1 fit.
   333→          </p>
   334→        </div>
   335→
   336→        {/* Live preview */}
   337→        <div className="hidden flex-col items-center rounded-xl border bg-stone-100/70 p-3 sm:flex" aria-hidden>
   338→          <span className="mb-2 text-[10px] font-semibold uppercase tracking-wider text-stone-400">Preview</span>
   339→          <PaperPreview sheet={sheet} paddingMm={paddingMm} fontSize={fontSize}>
   340→            <DocBody branding={branding} data={data} mono={sheet.mono} />
   341→          </PaperPreview>
   342→        </div>
   343→      </div>
   344→
   345→      {data.reprint ? (
   346→        <div className="no-print">
   347→          <span className="rounded border border-amber-300 bg-amber-50 px-2 py-1 text-xs font-medium text-amber-700">REPRINT</span>
   348→        </div>
   349→      ) : null}
   350→
   351→      {/* Actual printable document */}
   352→      <div
   353→        className={`print-area mx-auto bg-white text-stone-900 shadow-sm ${sheet.mono ? "font-mono" : "font-sans"}`}
   354→        style={
   355→          {
   356→            "--pp-width": `${fmtMm(sheet.widthMm)}mm`,
   357→            width: `${fmtMm(sheet.widthMm)}mm`,
   358→            maxWidth: "100%",
   359→            padding: `${fmtMm(paddingMm)}mm`,
   360→            fontSize: `${fontSize}px`,
   361→            lineHeight: 1.45,
   362→          } as React.CSSProperties
   363→        }
   364→      >
   365→        <DocBody branding={branding} data={data} mono={sheet.mono} />
   366→      </div>
   367→    </div>
   368→  );
   369→}

   371→/** Scaled paper mock: true aspect ratio, live doc render inside. */
   372→function PaperPreview({
   373→  sheet,
   374→  paddingMm,
   375→  fontSize,
   376→  children,
   377→}: {
   378→  sheet: { widthMm: number; heightMm: number | null };
   379→  paddingMm: number;
   380→  fontSize: number;
   381→  children: ReactNode;
   382→}) {
   383→  const PREVIEW_W = 172; // px
   384→  const ratio = sheet.heightMm === null ? 1.5 : Math.min(1.9, sheet.heightMm / sheet.widthMm);
   385→  const zoom = PREVIEW_W / (sheet.widthMm * PX_PER_MM); // doc laid out at true mm size, zoomed to fit
   386→  return (
   387→    <div
   388→      className="relative overflow-hidden rounded-md border border-stone-300 bg-white shadow-sm"
   389→      style={{ width: PREVIEW_W, height: Math.round(PREVIEW_W * ratio) }}
   390→    >
   391→      <div
   392→        style={{
   393→          width: `${sheet.widthMm}mm`,
   394→          padding: `${paddingMm}mm`,
   395→          fontSize: `${fontSize}px`,
   396→          zoom,
   397→          transformOrigin: "top left",
   398→        }}
   399→      >
   400→        {children}
   401→      </div>
   402→    </div>
   403→  );
   404→}

   406→function DocBody({ branding, data, mono }: { branding: Branding; data: ReceiptDocData; mono: boolean }) {
   407→  const money = (m: number) => {
   408→    try {
   409→      return new Intl.NumberFormat("en-NG", { style: "currency", currency: branding.currency }).format(m / 100);
   410→    } catch {
   411→      return `${branding.currency} ${(m / 100).toFixed(2)}`;
   412→    }
   413→  };
   414→  const isInvoice = data.kind === "INVOICE";
   415→  const isCredit = data.kind === "CREDIT_NOTE";
   416→  const docTitle = isInvoice ? "INVOICE" : isCredit ? "CREDIT NOTE" : "OFFICIAL RECEIPT";
   417→  const center = "text-center";
   418→  const gap = mono ? "my-1.5 border-t border-dashed border-stone-400" : "my-2 border-t border-stone-200";
   419→
   420→  return (
   421→    <div>
   422→      <div className={center}>
   423→        {branding.logoData ? (
   424→          <img
   425→            src={branding.logoData}
   426→            alt={`${branding.facilityName} logo`}
   427→            className={`mx-auto mb-1 ${mono ? "h-8" : "h-14"} object-contain`}
   428→          />
   429→        ) : null}
   430→        <div className={mono ? "text-[11px] font-bold uppercase" : "text-base font-bold uppercase tracking-wide"}>
   431→          {branding.facilityName}
   432→        </div>
   433→        <div className="text-[10px] text-stone-600">
   434→          {branding.address ? <div>{branding.address}</div> : null}
   435→          {branding.phone ? <div>Tel: {branding.phone}</div> : null}
   436→        </div>
   437→        <div className="mt-1 font-semibold">{docTitle}</div>
   438→      </div>
   439→
   440→      <div className={gap} />
   441→
   442→      <div className="grid grid-cols-2 gap-x-2 text-[10px]">
   443→        <div>No: <span className="font-semibold">{data.number}</span></div>
   444→        <div className="text-right">{data.dateTime}</div>
   445→        <div className="col-span-2">{isCredit ? "Credit for" : "Patient/Customer"}: <span className="font-semibold">{data.issuedTo}</span></div>
   446→        <div className="col-span-2">Issued by: {data.cashier}</div>
   447→      </div>
   448→
   449→      <div className={gap} />
   450→
   451→      <table className="w-full text-[10px]">
   452→        <thead>
   453→          <tr className="text-left">
   454→            <th className="pb-1">Item</th>
   455→            <th className="pb-1 text-right">Qty</th>
   456→            <th className="pb-1 text-right">Rate</th>
   457→            <th className="pb-1 text-right">Amt</th>
   458→          </tr>
   459→        </thead>
   460→        <tbody>
   461→          {data.lines.map((line, i) => (
   462→            <tr key={i} className="align-top break-inside-avoid">
   463→              <td className="pb-1 pr-1">{line.name}</td>
   464→              <td className="pb-1 text-right">{line.quantity}</td>
   465→              <td className="pb-1 text-right">{money(line.unitPriceMinor)}</td>
   466→              <td className="pb-1 text-right">{money(line.totalMinor)}</td>
   467→            </tr>
   468→          ))}
   469→        </tbody>
   470→      </table>
   471→
   472→      <div className={gap} />
   473→
   474→      <div className="space-y-0.5 text-[10px]">
   475→        <div className="flex justify-between"><span>Subtotal</span><span>{money(data.subtotalMinor)}</span></div>
   476→        {data.discountMinor > 0 ? (
   477→          <div className="flex justify-between"><span>Discount</span><span>-{money(data.discountMinor)}</span></div>
   478→        ) : null}
   479→        <div className="flex justify-between text-[11px] font-bold">
   480→          <span>{isInvoice ? "Total due" : isCredit ? "Refund total" : "Total"}</span>
   481→          <span>{money(data.totalMinor)}</span>
   482→        </div>
   483→        {!isInvoice ? (
   484→          <>
   485→            <div className="flex justify-between"><span>{isCredit ? "Refund method" : "Paid"} ({data.method})</span><span>{money(data.amountPaidMinor ?? data.totalMinor)}</span></div>
   486→            {data.reference ? <div className="flex justify-between"><span>Ref</span><span className="truncate pl-2">{data.reference}</span></div> : null}
   487→          </>
   488→        ) : (
   489→          <div className="flex justify-between"><span>Amount paid</span><span>{money(data.amountPaidMinor ?? 0)}</span></div>
   490→        )}
   491→      </div>
   492→
   493→      <div className={gap} />
   494→
   495→      <div className={`${center} text-[9px] text-stone-600`}>
   496→        {data.reprint ? <div className="mb-1 font-bold text-stone-800">** REPRINT **</div> : null}
   497→        <div>{isInvoice ? branding.reportFooter ?? "Thank you." : branding.receiptFooter ?? "Thank you."}</div>
   498→        <div className="mt-1">{branding.facilityName} · {data.number}</div>
   499→      </div>
   500→    </div>
   501→  );
   502→}

   504→/* ─── Page-level print: any paper size for full-page reports ─────────────── */
   505→
   506→const PAGE_PRINT_KEY = "curavilo.pageprint.v1";
   507→
   508→interface PagePrintSettings {
   509→  profileKey: PrintProfileKey;
   510→  orientation: Orientation;
   511→  customWidthMm: number;
   512→  customHeightMm: number;
   513→  customUnit: Unit;
   514→}
   515→
   516→const PAGE_PRINT_DEFAULTS: PagePrintSettings = {
   517→  profileKey: "A4_STANDARD",
   518→  orientation: "portrait",
   519→  customWidthMm: 210,
   520→  customHeightMm: 297,
   521→  customUnit: "mm",
   522→};
   523→
   524→export function PagePrintButton({ label = "Print", className }: { label?: string; className?: string }) {
   525→  const [cfg, setCfg] = useState<PagePrintSettings>(() => {
   526→    if (typeof window === "undefined") return PAGE_PRINT_DEFAULTS;
   527→    try {
   528→      const raw = window.localStorage.getItem(PAGE_PRINT_KEY);
   529→      return raw ? { ...PAGE_PRINT_DEFAULTS, ...(JSON.parse(raw) as Partial<PagePrintSettings>) } : PAGE_PRINT_DEFAULTS;
   530→    } catch {
   531→      return PAGE_PRINT_DEFAULTS;
   532→    }
   533→  });
   534→
   535→  useEffect(() => {
   536→    try {
   537→      window.localStorage.setItem(PAGE_PRINT_KEY, JSON.stringify(cfg));
   538→    } catch {
   539→      // storage unavailable
   540→    }
   541→    const sheet = resolveSheet({ ...cfg, marginMm: 0, scalePct: 100 });
   542→    const size = sheet.heightMm === null ? `${fmtMm(sheet.widthMm)}mm auto` : `${fmtMm(sheet.widthMm)}mm ${fmtMm(sheet.heightMm)}mm`;
   543→    let style = document.querySelector<HTMLStyleElement>("style[data-page-print]");
   544→    if (!style) {
   545→      style = document.createElement("style");
   546→      style.setAttribute("data-page-print", "true");
   547→      document.head.appendChild(style);
   548→    }
   549→    style.textContent = `@media print { @page { size: ${size}; margin: 8mm; } }`;
   550→    return () => {
   551→      style.textContent = "";
   552→    };
   553→  }, [cfg]);
   554→
   555→  const update = (patch: Partial<PagePrintSettings>) => setCfg((prev) => ({ ...prev, ...patch }));
   556→  const thermal = isThermal(cfg.profileKey);
   557→
   558→  return (
   559→    <div className="flex items-center gap-1.5 no-print">
   560→      <Popover>
   561→        <PopoverTrigger asChild>
   562→          <Button variant="outline" size="sm" className={className} aria-label="Page print settings">
   563→            <Settings2 className="h-4 w-4" />
   564→            <span className="sr-only">Page print settings</span>
   565→          </Button>
   566→        </PopoverTrigger>
   567→        <PopoverContent align="end" className="w-72 space-y-2.5">
   568→          <div className="space-y-1">
   569→            <Label className="text-[11px] text-muted-foreground">Paper size</Label>
   570→            <Select value={cfg.profileKey} onValueChange={(v) => update({ profileKey: v as PrintProfileKey })}>
   571→              <SelectTrigger aria-label="Page paper size"><SelectValue /></SelectTrigger>
   572→              <SelectContent>
   573→                {Object.values(PRINT_PROFILES).map((p) => (
   574→                  <SelectItem key={p.key} value={p.key}>{p.label}</SelectItem>
   575→                ))}
   576→                <SelectItem value="CUSTOM">Custom size…</SelectItem>
   577→              </SelectContent>
   578→            </Select>
   579→          </div>
   580→          {cfg.profileKey === "CUSTOM" ? (
   581→            <div className="flex items-end gap-2">
   582→              <div className="flex-1 space-y-1">
   583→                <Label className="text-[11px] text-muted-foreground">Width</Label>
   584→                <Input
   585→                  type="number"
   586→                  min={10}
   587→                  value={cfg.customUnit === "mm" ? cfg.customWidthMm : Math.round((cfg.customWidthMm / MM_PER_IN) * 100) / 100}
   588→                  onChange={(e) => update({ customWidthMm: Math.max(10, Number(e.target.value) || 0) * (cfg.customUnit === "in" ? MM_PER_IN : 1) })}
   589→                />
   590→              </div>
   591→              <div className="flex-1 space-y-1">
   592→                <Label className="text-[11px] text-muted-foreground">Height</Label>
   593→                <Input
   594→                  type="number"
   595→                  min={10}
   596→                  value={cfg.customUnit === "mm" ? cfg.customHeightMm : Math.round((cfg.customHeightMm / MM_PER_IN) * 100) / 100}
   597→                  onChange={(e) => update({ customHeightMm: Math.max(10, Number(e.target.value) || 0) * (cfg.customUnit === "in" ? MM_PER_IN : 1) })}
   598→                />
   599→              </div>
   600→              <Select value={cfg.customUnit} onValueChange={(v) => update({ customUnit: v as Unit })}>
   601→                <SelectTrigger className="w-[70px]" aria-label="Unit"><SelectValue /></SelectTrigger>
   602→                <SelectContent>
   603→                  <SelectItem value="mm">mm</SelectItem>
   604→                  <SelectItem value="in">inch</SelectItem>
   605→                </SelectContent>
   606→              </Select>
   607→            </div>
   608→          ) : null}
   609→          {!thermal ? (
   610→            <div className="space-y-1">
   611→              <Label className="text-[11px] text-muted-foreground">Orientation</Label>
   612→              <div className="flex h-9 overflow-hidden rounded-md border">
   613→                {(["portrait", "landscape"] as Orientation[]).map((o) => (
   614→                  <button
   615→                    key={o}
   616→                    type="button"
   617→                    onClick={() => update({ orientation: o })}
   618→                    className={`flex-1 text-xs transition ${cfg.orientation === o ? "bg-teal-600 font-medium text-white" : "bg-white text-stone-500 hover:bg-teal-50"}`}
   619→                  >
   620→                    {o === "portrait" ? "Portrait" : "Landscape"}
   621→                  </button>
   622→                ))}
   623→              </div>
   624→            </div>
   625→          ) : null}
   626→          <p className="text-[11px] text-muted-foreground">Applies to this page when you print. Your choice is remembered.</p>
   627→        </PopoverContent>
   628→      </Popover>
   629→      <Button variant="outline" size="sm" onClick={() => window.print()} className={className}>
   630→        <Printer className="mr-2 h-4 w-4" /> {label}
   631→      </Button>
   632→    </div>
   633→  );
   634→}
