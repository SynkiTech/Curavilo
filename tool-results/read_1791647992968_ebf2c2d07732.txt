     8→function hashPassword(password: string): string {
     9→  const salt = randomBytes(16).toString("hex");
    10→  return `${salt}:${scryptSync(password, salt, 64).toString("hex")}`;
    11→}

    15→function daysAgo(n: number, hour = 9): Date {
    16→  const d = new Date();
    17→  d.setUTCHours(hour, 0, 0, 0);
    18→  d.setUTCDate(d.getUTCDate() - n);
    19→  return d;
    20→}

    22→function salePeriodForDays(n: number): string {
