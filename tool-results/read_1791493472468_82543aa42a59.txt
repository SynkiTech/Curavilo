   194→
   195→function safeParse(def: string): BuilderInitial | undefined {
   196→  try {
   197→    const parsed = JSON.parse(def) as { trigger?: string; nodes?: Array<{ id: string; type: string; config: Record<string, unknown> }>; schedule?: { everyMinutes?: number; dailyAt?: string } };
   198→    return {
   199→      definition: {
   200→        ...parsed,
   201→        nodes: parsed.nodes?.map((n) => ({ id: n.id, type: n.type as BuilderNode["type"], config: Object.fromEntries(Object.entries(n.config ?? {}).map(([k, v]) => [k, String(v)])) })),
   202→      },
   203→    };
   204→  } catch {
   205→    return undefined;
   206→  }
   207→}

   389→                      </Button>
   390→                    </span>
   391→                  ) : null}
   392→                </li>
   393→              );
   394→            })}
   395→          </ul>
   396→        )
   397→      ) : null}

   399→  );

   666→                        <TableHead className="w-24 text-right">Actions</TableHead>
   667→                      </TableRow>

   687→                              </Button>
   688→                            </div>
   689→                          </TableCell>
   690→                        </TableRow>
   691→                      ))}
   692→                    </TableBody>
   693→                  </Table>
   694→                </div>

   697→          </Card>
   698→          {can("org:manage") ? <FacilityCredentialCard /> : null}
   699→        </div>
   700→      </div>
   701→
   702→      {/* Profile switch confirmation */}

   705→          <AlertDialogHeader>

   707→            <AlertDialogDescription>
   708→              {pendingProfile ? data.profiles.find((p) => p.key === pendingProfile)?.name : ""} changes the default terminology and the encounter form template for this facility. Existing records keep their data, and the service catalogue is not modified. You can customise terminology afterwards.
   709→            </AlertDialogDescription>
   710→          </AlertDialogHeader>
   711→          <AlertDialogFooter>
   712→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   713→            <AlertDialogAction className="bg-teal-700 hover:bg-teal-800" onClick={() => void confirmProfileSwitch()}>Switch profile</AlertDialogAction>
   714→          </AlertDialogFooter>
   715→        </AlertDialogContent>
   716→      </AlertDialog>
   717→
   718→      {/* Add / edit service dialog */}
   719→      {serviceDialog ? (
   720→        <ServiceDialog
   721→          mode={serviceDialog.mode}
   722→          service={serviceDialog.mode === "edit" ? serviceDialog.service : null}
   723→          currency={currency}
   724→          onClose={() => setServiceDialog(null)}
   725→          onSaved={async () => {
   726→            setServiceDialog(null);
   727→            void services.refetch();
   728→          }}
   729→        />
   730→      ) : null}
   731→    </div>
   732→  );
   733→}

   735→// ─── Service catalogue add/edit dialog ──────────────────────────────────────
   736→
   737→function ServiceDialog({ mode, service, currency, onClose, onSaved }: {
   738→  mode: "add" | "edit";
   739→  service: ServiceItem | null;
   740→  currency: string;
   741→  onClose: () => void;
   742→  onSaved: () => void | Promise<void>;
   743→}) {
   744→  const [code, setCode] = useState(service?.code ?? "");
   745→  const [name, setName] = useState(service?.name ?? "");
   746→  const [category, setCategory] = useState<string>(service?.category ?? "CONSULTATION");
   747→  const [price, setPrice] = useState(service ? (service.priceMinor / 100).toFixed(2) : "");
   748→  const [busy, setBusy] = useState(false);
   749→  const [err, setErr] = useState<string | null>(null);
   750→
   751→  const submit = async () => {
   752→    setErr(null);
   753→    const priceMinor = Math.round(parseFloat(price) * 100);
   754→    if (!code.trim() || code.trim().length < 2) { setErr("Code is required (min 2 characters)."); return; }
   755→    if (!name.trim() || name.trim().length < 2) { setErr("Name is required (min 2 characters)."); return; }
   756→    if (!Number.isFinite(priceMinor) || priceMinor < 0) { setErr("Enter a valid price (0 or more)."); return; }
   757→    setBusy(true);
   758→    try {
   759→      if (mode === "add") {
   760→        await apiSend("POST", "/api/services", { code: code.trim(), name: name.trim(), category, priceMinor });
   761→        toast.success(`Service "${name.trim()}" added to the catalogue.`);
   762→      } else if (service) {
   763→        await apiSend("PATCH", "/api/services", { id: service.id, code: code.trim(), name: name.trim(), category, priceMinor });
   764→        toast.success(`Service "${name.trim()}" updated - the new price applies facility-wide.`);
   765→      }
   766→      await onSaved();
   767→      onClose();
   768→    } catch (e) {
   769→      setErr((e as Error).message);
   770→    } finally {
   771→      setBusy(false);
   772→    }
   773→  };
   774→
   775→  return (
   776→    <Dialog open onOpenChange={(open) => !open && onClose()}>
   777→      <DialogContent className="sm:max-w-md">
   778→        <DialogHeader>
   779→          <DialogTitle>{mode === "add" ? "Add service" : "Edit service"}</DialogTitle>
   780→          <DialogDescription>
   781→            {mode === "add" ? "New services become available for booking, encounters, billing and the POS." : "Price and detail changes apply across the whole facility immediately."}
   782→          </DialogDescription>
   783→        </DialogHeader>
   784→        <div className="space-y-3">
   785→          {err ? <p className="rounded-md border border-red-200 bg-red-50 px-3 py-2 text-xs text-red-700" role="alert">{err}</p> : null}
   786→          <div className="grid grid-cols-2 gap-3">
   787→            <div className="space-y-1">
   788→              <Label htmlFor="svc-code">Code</Label>
   789→              <Input id="svc-code" value={code} maxLength={40} onChange={(e) => setCode(e.target.value.toUpperCase())} placeholder="e.g. CONS-EYE" />
   790→            </div>
   791→            <div className="space-y-1">
   792→              <Label htmlFor="svc-cat">Category</Label>
   793→              <Select value={category} onValueChange={setCategory}>
   794→                <SelectTrigger id="svc-cat" aria-label="Category"><SelectValue /></SelectTrigger>
   795→                <SelectContent>
   796→                  {SERVICE_CATEGORIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   797→                </SelectContent>
   798→              </Select>
   799→            </div>
   800→          </div>
   801→          <div className="space-y-1">
   802→            <Label htmlFor="svc-name">Service name</Label>
   803→            <Input id="svc-name" value={name} maxLength={120} onChange={(e) => setName(e.target.value)} placeholder="e.g. Eye Consultation" />
   804→          </div>
   805→          <div className="space-y-1">
   806→            <Label htmlFor="svc-price">Price ({currency})</Label>
   807→            <Input id="svc-price" inputMode="decimal" value={price} onChange={(e) => setPrice(e.target.value)} placeholder="0.00" />
   808→            {mode === "edit" && service ? (
   809→              <p className="text-[11px] text-muted-foreground">Current: {formatMoney(service.priceMinor, service.currency)}</p>
   810→            ) : null}
   811→          </div>
   812→        </div>
   813→        <DialogFooter>
   814→          <Button variant="outline" onClick={onClose} disabled={busy}>Cancel</Button>
   815→          <Button className="bg-teal-700 hover:bg-teal-800" onClick={() => void submit()} disabled={busy}>
   816→            {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null}
   817→            {mode === "add" ? "Add service" : "Save changes"}
   818→          </Button>
   819→        </DialogFooter>
   820→      </DialogContent>
   821→    </Dialog>
   822→  );
   823→}

   825→// ─── Facility access credential (owner-only, ADR 0011) ─────────────────────
   826→
   827→function FacilityCredentialCard() {
   828→  const [email, setEmail] = useState("");
   829→  const [password, setPassword] = useState("");
   830→  const [confirm, setConfirm] = useState("");
   831→  const [busy, setBusy] = useState(false);
   832→  const [loading, setLoading] = useState(true);
   833→
   834→  useEffect(() => {
   835→    let active = true;
   836→    apiGet<{ credentialEmail: string | null }>("/api/facility-credential")
   837→      .then((r) => { if (active) { setEmail(r.credentialEmail ?? ""); setLoading(false); } })
   838→      .catch(() => active && setLoading(false));
   839→    return () => { active = false; };
   840→  }, []);
   841→
   842→  const save = async () => {
   843→    if (password && password !== confirm) {
   844→      toast.error("New passwords do not match.");
   845→      return;
   846→    }
   847→    setBusy(true);
   848→    try {
   849→      await apiSend("PATCH", "/api/facility-credential", {
   850→        ...(email ? { email } : {}),
   851→        ...(password ? { password } : {}),
   852→      });
   853→      toast.success("Facility access credential updated.");
   854→      setPassword("");
   855→      setConfirm("");
   856→    } catch (e) {
   857→      toast.error((e as Error).message);
   858→    } finally {
   859→      setBusy(false);
   860→    }
   861→  };
   862→
   863→  return (
   864→    <Card>
   865→      <CardHeader><CardTitle className="text-sm">Facility access credential</CardTitle></CardHeader>
   866→      <CardContent className="space-y-3">
   867→        <p className="text-xs text-muted-foreground">
   868→          This is the workspace credential used at the facility access gate (step 1 of sign-in). It opens
   869→          the facility only - it never grants staff permissions. Owner-only, changeable at any time.
   870→        </p>
   871→        {loading ? <LoadingState /> : (
   872→          <>
   873→            <div className="space-y-1">
   874→              <Label htmlFor="cred-email">Facility gate email</Label>
   875→              <Input id="cred-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
   876→            </div>
   877→            <div className="grid grid-cols-2 gap-3">
   878→              <div className="space-y-1">
