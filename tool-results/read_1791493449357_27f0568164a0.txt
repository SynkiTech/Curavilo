   118→   118→function loadSettings(): PrintSettings {
   119→   119→  try {
   120→   120→    const raw = window.localStorage.getItem(STORAGE_KEY);
   121→   121→    if (!raw) return DEFAULTS;
   122→   122→    const parsed = JSON.parse(raw) as Partial<PrintSettings>;
   123→   123→    return { ...DEFAULTS, ...parsed };
   124→   124→  } catch {
   125→   125→    return DEFAULTS;
   126→   126→  }
   127→   127→}

   129→   129→const isThermal = (key: PrintProfileKey) => key === "THERMAL_58MM" || key === "THERMAL_80MM" || key === "THERMAL_112MM";

   131→   131→/** Resolve the effective sheet geometry from settings. */
   132→   132→function resolveSheet(s: PrintSettings): { widthMm: number; heightMm: number | null; mono: boolean } {
   133→   133→  if (s.profileKey === "CUSTOM") {
   134→   134→    const w = s.customUnit === "in" ? s.customWidthMm * MM_PER_IN : s.customWidthMm;
   135→   135→    const h = s.customUnit === "in" ? s.customHeightMm * MM_PER_IN : s.customHeightMm;
   136→   136→    const [ew, eh] = s.orientation === "landscape" ? [Math.max(w, h), Math.min(w, h)] : [w, h];
   137→   137→    return { widthMm: ew, heightMm: eh, mono: false };
   138→   138→  }
   139→   139→  const p = PRINT_PROFILES[s.profileKey as Exclude<PrintProfileKey, "CUSTOM">];
   140→   140→  if (p.heightMm === null) return { widthMm: p.widthMm, heightMm: null, mono: p.mono };
   141→   141→  const [ew, eh] = s.orientation === "landscape" ? [p.heightMm, p.widthMm] : [p.widthMm, p.heightMm];
   142→   142→  return { widthMm: ew, heightMm: eh, mono: p.mono };
   143→   143→}

   229→   229→                    <SelectItem key={p.key} value={p.key}>{p.label}</SelectItem>

   351→   351→      {/* Actual printable document */}

   371→   371→/** Scaled paper mock: true aspect ratio, live doc render inside. */
   372→   372→function PaperPreview({
   373→   373→  sheet,
   374→   374→  paddingMm,
   375→   375→  fontSize,
   376→   376→  children,
   377→   377→}: {
   378→   378→  sheet: { widthMm: number; heightMm: number | null };
   379→   379→  paddingMm: number;
   380→   380→  fontSize: number;
   381→   381→  children: ReactNode;
   382→   382→}) {
   383→   383→  const PREVIEW_W = 172; // px
   384→   384→  const ratio = sheet.heightMm === null ? 1.5 : Math.min(1.9, sheet.heightMm / sheet.widthMm);
   385→   385→  const zoom = PREVIEW_W / (sheet.widthMm * PX_PER_MM); // doc laid out at true mm size, zoomed to fit
   386→   386→  return (
   387→   387→    <div
   388→   388→      className="relative overflow-hidden rounded-md border border-stone-300 bg-white shadow-sm"
   389→   389→      style={{ width: PREVIEW_W, height: Math.round(PREVIEW_W * ratio) }}
   390→   390→    >
   391→   391→      <div
   392→   392→        style={{
   393→   393→          width: `${sheet.widthMm}mm`,
   394→   394→          padding: `${paddingMm}mm`,
   395→   395→          fontSize: `${fontSize}px`,
   396→   396→          zoom,
   397→   397→          transformOrigin: "top left",
   398→   398→        }}
   399→   399→      >
   400→   400→        {children}
   401→   401→      </div>
   402→   402→    </div>
   403→   403→  );
   404→   404→}

   406→   406→function DocBody({ branding, data, mono }: { branding: Branding; data: ReceiptDocData; mono: boolean }) {
   407→   407→  const money = (m: number) => {
   408→   408→    try {
   409→   409→      return new Intl.NumberFormat("en-NG", { style: "currency", currency: branding.currency }).format(m / 100);
   410→   410→    } catch {
   411→   411→      return `${branding.currency} ${(m / 100).toFixed(2)}`;
   412→   412→    }
   413→   413→  };
   414→   414→  const isInvoice = data.kind === "INVOICE";
   415→   415→  const isCredit = data.kind === "CREDIT_NOTE";

   506→   506→const PAGE_PRINT_KEY = "curavilo.pageprint.v1";
   507→   507→
   508→   508→interface PagePrintSettings {
   509→   509→  profileKey: PrintProfileKey;
   510→   510→  orientation: Orientation;
   511→   511→  customWidthMm: number;
   512→   512→  customHeightMm: number;
   513→   513→  customUnit: Unit;
   514→   514→}
   515→   515→
   516→   516→const PAGE_PRINT_DEFAULTS: PagePrintSettings = {
   517→   517→  profileKey: "A4_STANDARD",
   518→   518→  orientation: "portrait",
   519→   519→  customWidthMm: 210,
   520→   520→  customHeightMm: 297,
   521→   521→  customUnit: "mm",
   522→   522→};
   523→   523→
   524→   524→export function PagePrintButton({ label = "Print", className }: { label?: string; className?: string }) {
   525→   525→  const [cfg, setCfg] = useState<PagePrintSettings>(() => {
   526→   526→    if (typeof window === "undefined") return PAGE_PRINT_DEFAULTS;
   527→   527→    try {
   528→   528→      const raw = window.localStorage.getItem(PAGE_PRINT_KEY);
   529→   529→      return raw ? { ...PAGE_PRINT_DEFAULTS, ...(JSON.parse(raw) as Partial<PagePrintSettings>) } : PAGE_PRINT_DEFAULTS;
   530→   530→    } catch {
   531→   531→      return PAGE_PRINT_DEFAULTS;
   532→   532→    }
   533→   533→  });
   534→   534→
   535→   535→  useEffect(() => {
   536→   536→    try {
   537→   537→      window.localStorage.setItem(PAGE_PRINT_KEY, JSON.stringify(cfg));
   538→   538→    } catch {
   539→   539→      // storage unavailable
   540→   540→    }
   541→   541→    const sheet = resolveSheet({ ...cfg, marginMm: 0, scalePct: 100 });
   542→   542→    const size = sheet.heightMm === null ? `${fmtMm(sheet.widthMm)}mm auto` : `${fmtMm(sheet.widthMm)}mm ${fmtMm(sheet.heightMm)}mm`;
   543→   543→    let style = document.querySelector<HTMLStyleElement>("style[data-page-print]");
   544→   544→    if (!style) {
   545→   545→      style = document.createElement("style");
   546→   546→      style.setAttribute("data-page-print", "true");
   547→   547→      document.head.appendChild(style);
   548→   548→    }
   549→   549→    style.textContent = `@media print { @page { size: ${size}; margin: 8mm; } }`;
   550→   550→    return () => {
   551→   551→      style.textContent = "";
   552→   552→    };
   553→   553→  }, [cfg]);
   554→   554→
   555→   555→  const update = (patch: Partial<PagePrintSettings>) => setCfg((prev) => ({ ...prev, ...patch }));
   556→   556→  const thermal = isThermal(cfg.profileKey);
   557→   557→
   558→   558→  return (
   559→   559→    <div className="flex items-center gap-1.5 no-print">
   560→   560→      <Popover>

   569→   569→            <Label className="text-[11px] text-muted-foreground">Paper size</Label>
   570→   570→            <Select value={cfg.profileKey} onValueChange={(v) => update({ profileKey: v as PrintProfileKey })}>
   571→   571→              <SelectTrigger aria-label="Page paper size"><SelectValue /></SelectTrigger>

   573→   573→                {Object.values(PRINT_PROFILES).map((p) => (
