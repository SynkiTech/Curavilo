   192→
   193→function safeParse(def: string): BuilderInitial | undefined {
   194→  try {
   195→    const parsed = JSON.parse(def) as { trigger?: string; nodes?: Array<{ id: string; type: string; config: Record<string, unknown> }>; schedule?: { everyMinutes?: number; dailyAt?: string } };
   196→    return {
   197→      definition: {
   198→        ...parsed,
   199→        nodes: parsed.nodes?.map((n) => ({ id: n.id, type: n.type as BuilderNode["type"], config: Object.fromEntries(Object.entries(n.config ?? {}).map(([k, v]) => [k, String(v)])) })),
   200→      },
   201→    };
   202→  } catch {
   203→    return undefined;
   204→  }
   205→}

   392→          </ul>

   423→
   424→  if (isLoading) return <LoadingState />;
   425→  if (isError) return <ErrorState message={(error as Error)?.message} onRetry={() => void refetch()} />;
   426→  if (!data) return null;
   427→
   428→  const f = { ...data.facility, ...form };
   429→  const currentLogo = logoPreview ?? f.logoData ?? null;
   430→
   431→  const pickLogo = (file: File | undefined) => {
   432→    if (!file) return;
   433→    if (!/^image\/(png|jpe?g|webp)$/.test(file.type)) { toast.error("Logo must be PNG, JPEG or WebP."); return; }
   434→    if (file.size > 280_000) { toast.error("Logo must be smaller than 280 KB."); return; }
   435→    const reader = new FileReader();
   436→    reader.onload = () => setLogoPreview(String(reader.result));
   437→    reader.readAsDataURL(file);
   438→  };
   439→
   440→  const save = async () => {
   441→    setBusy(true);
   442→    try {
   443→      await apiSend("PATCH", "/api/settings", {
   444→        name: f.name,
   445→        address: f.address ?? null,
   446→        phone: f.phone ?? null,
   447→        email: f.email ?? "",
   448→        logoData: currentLogo ?? "",
   449→        reportFooter: f.reportFooter ?? null,
   450→        receiptFooter: f.receiptFooter ?? null,
   451→      });
   452→      toast.success("Settings saved.");
   453→      setLogoPreview(null);
   454→      const fresh = await apiGet<SessionInfo>("/api/auth");
   455→      setSession(fresh ?? { authenticated: false });
   456→      refetch();
   457→    } catch (e) {
   458→      toast.error((e as Error).message);
   459→    } finally {
   460→      setBusy(false);
   461→    }
   462→  };
   463→
   464→  return (
   465→    <div>
   466→      <PageHeader title="Settings" subtitle="Facility profile, branding, terminology and service catalogue." />
   467→      <div className="grid gap-4 lg:grid-cols-2">
   468→        <Card>
   469→          <CardHeader><CardTitle className="text-sm">Facility profile & branding</CardTitle></CardHeader>
   470→          <CardContent className="space-y-3">
   471→            <div className="flex items-center gap-4">
   472→              <div className="flex h-16 w-16 items-center justify-center overflow-hidden rounded-lg border border-dashed bg-stone-50">
   473→                {currentLogo ? (
   474→                   
   475→                  <img src={currentLogo} alt="Facility logo" className="h-full w-full object-contain" />
   476→                ) : (
   477→                  <span className="text-xs text-stone-400">No logo</span>
   478→                )}
   479→              </div>
   480→              <div>
   481→                <input
   482→                  type="file"
   483→                  accept="image/png,image/jpeg,image/webp"
   484→                  className="hidden"
   485→                  ref={(el) => { fileRef[0] = el; }}
   486→                  onChange={(e) => pickLogo(e.target.files?.[0])}
   487→                />
   488→                <Button variant="outline" size="sm" onClick={() => fileRef[0]?.click()}>Upload logo</Button>
   489→                <p className="mt-1 text-[11px] text-muted-foreground">PNG/JPEG/WebP, max 280 KB. SVG rejected for safety.</p>
   490→              </div>
   491→            </div>
   492→            <div className="space-y-1">
   493→              <Label htmlFor="set-name">Facility name</Label>
   494→              <Input id="set-name" value={f.name ?? ""} onChange={(e) => setForm({ ...form, name: e.target.value })} />
   495→            </div>
   496→            <div className="space-y-1">
   497→              <Label htmlFor="set-address">Address</Label>
   498→              <Input id="set-address" value={f.address ?? ""} onChange={(e) => setForm({ ...form, address: e.target.value })} />
   499→            </div>
   500→            <div className="grid grid-cols-2 gap-3">
   501→              <div className="space-y-1">
   502→                <Label htmlFor="set-phone">Phone</Label>
   503→                <Input id="set-phone" value={f.phone ?? ""} onChange={(e) => setForm({ ...form, phone: e.target.value })} />
   504→              </div>
   505→              <div className="space-y-1">
   506→                <Label htmlFor="set-email">Email</Label>
   507→                <Input id="set-email" value={f.email ?? ""} onChange={(e) => setForm({ ...form, email: e.target.value })} />
   508→              </div>
   509→            </div>
   510→            <div className="space-y-1">
   511→              <Label htmlFor="set-rf">Report footer</Label>
   512→              <Input id="set-rf" value={f.reportFooter ?? ""} onChange={(e) => setForm({ ...form, reportFooter: e.target.value })} />
   513→            </div>
   514→            <div className="space-y-1">
   515→              <Label htmlFor="set-rf2">Receipt footer</Label>
   516→              <Input id="set-rf2" value={f.receiptFooter ?? ""} onChange={(e) => setForm({ ...form, receiptFooter: e.target.value })} />
   517→            </div>
   518→            <Button className="w-full bg-teal-700 hover:bg-teal-800" disabled={busy} onClick={() => void save()}>
   519→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null} Save settings
   520→            </Button>
   521→          </CardContent>
   522→        </Card>
   523→
   524→        <div className="space-y-4">
   525→          <Card>
   526→            <CardHeader><CardTitle className="text-sm">Service profile & terminology</CardTitle></CardHeader>
   527→            <CardContent>
   528→              <p className="text-sm font-medium">{data.profile.name}</p>
   529→              <p className="text-xs text-muted-foreground">{data.profile.description}</p>
   530→              <ul className="mt-3 space-y-1 text-sm">
   531→                {Object.entries(data.profile.terminology).map(([k, v]) => (
   532→                  <li key={k} className="flex justify-between rounded border px-2.5 py-1 text-xs">
   533→                    <span className="font-mono text-muted-foreground">{k}</span>
   534→                    <span className="font-medium">{v}</span>
   535→                  </li>
   536→                ))}
   537→              </ul>
   538→            </CardContent>
   539→          </Card>
   540→
   541→          <Card>
   542→            <CardHeader><CardTitle className="text-sm">Service catalogue</CardTitle></CardHeader>
   543→            <CardContent>
   544→              <ul className="max-h-56 space-y-1 overflow-y-auto text-sm">
   545→                {(services.data?.services ?? []).map((s) => (
   546→                  <li key={s.id} className="flex items-center justify-between rounded border px-2.5 py-1.5">
   547→                    <span>
   548→                      {s.name}
   549→                      <span className="ml-2 font-mono text-[10px] text-muted-foreground">{s.code}</span>
   550→                    </span>
   551→                    <span className="tabular-nums">{formatMoney(s.priceMinor, s.currency)}</span>
   552→                  </li>
   553→                ))}
   554→              </ul>
   555→            </CardContent>
   556→          </Card>
   557→          {can("org:manage") ? <FacilityCredentialCard /> : null}
   558→        </div>
   559→      </div>
   560→    </div>
   561→  );
   562→}

   564→// ─── Facility access credential (owner-only, ADR 0011) ─────────────────────
   565→
   566→function FacilityCredentialCard() {
   567→  const [email, setEmail] = useState("");
   568→  const [password, setPassword] = useState("");
   569→  const [confirm, setConfirm] = useState("");
   570→  const [busy, setBusy] = useState(false);
   571→  const [loading, setLoading] = useState(true);
   572→
   573→  useEffect(() => {
   574→    let active = true;
   575→    apiGet<{ credentialEmail: string | null }>("/api/facility-credential")
   576→      .then((r) => { if (active) { setEmail(r.credentialEmail ?? ""); setLoading(false); } })
   577→      .catch(() => active && setLoading(false));
   578→    return () => { active = false; };
   579→  }, []);
   580→
   581→  const save = async () => {
   582→    if (password && password !== confirm) {
   583→      toast.error("New passwords do not match.");
   584→      return;
   585→    }
   586→    setBusy(true);
   587→    try {
   588→      await apiSend("PATCH", "/api/facility-credential", {
   589→        ...(email ? { email } : {}),
   590→        ...(password ? { password } : {}),
   591→      });
   592→      toast.success("Facility access credential updated.");
   593→      setPassword("");
   594→      setConfirm("");
   595→    } catch (e) {
   596→      toast.error((e as Error).message);
   597→    } finally {
   598→      setBusy(false);
   599→    }
   600→  };
   601→
   602→  return (
   603→    <Card>
   604→      <CardHeader><CardTitle className="text-sm">Facility access credential</CardTitle></CardHeader>
   605→      <CardContent className="space-y-3">
   606→        <p className="text-xs text-muted-foreground">
   607→          This is the workspace credential used at the facility access gate (step 1 of sign-in). It opens
   608→          the facility only - it never grants staff permissions. Owner-only, changeable at any time.
   609→        </p>
   610→        {loading ? <LoadingState /> : (
   611→          <>
   612→            <div className="space-y-1">
   613→              <Label htmlFor="cred-email">Facility gate email</Label>
   614→              <Input id="cred-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} />
   615→            </div>
   616→            <div className="grid grid-cols-2 gap-3">
   617→              <div className="space-y-1">
   618→                <Label htmlFor="cred-pass">New gate password</Label>
   619→                <Input id="cred-pass" type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder="Leave blank to keep" />
   620→              </div>
   621→              <div className="space-y-1">
   622→                <Label htmlFor="cred-confirm">Confirm password</Label>
   623→                <Input id="cred-confirm" type="password" value={confirm} onChange={(e) => setConfirm(e.target.value)} />
   624→              </div>
   625→            </div>
   626→            <Button className="w-full bg-teal-700 hover:bg-teal-800" disabled={busy || (!password && !email)} onClick={() => void save()}>
   627→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : null} Update facility credential
   628→            </Button>
   629→          </>
   630→        )}
   631→      </CardContent>
   632→    </Card>
   633→  );
   634→}
