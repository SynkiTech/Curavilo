    60→    60→/** Role preview for the staff creation form (spec §39): understandable can/cannot lists. */
    61→    61→export async function rolePreview(orgId: string, role: string) {
    62→    62→  let perms: string[];
    63→    63→  if (isBuiltInRole(role)) {
    64→    64→    perms = permissionsForRole(role);
    65→    65→  } else {
    66→    66→    const custom = await db.customRole.findFirst({ where: { orgId, name: role } });
    67→    67→    perms = custom ? JSON.parse(custom.permissions) as string[] : [];
    68→    68→  }
    69→    69→  const moduleOf: Record<string, string> = {
    70→    70→    "patients:read": "Patients", "patients:write": "Patient registration",
    71→    71→    "appointments:read": "Appointments", "appointments:write": "Appointment scheduling",
    72→    72→    "queue:operate": "Queue management", "encounters:read": "Clinical encounters",
    73→    73→    "encounters:write": "Encounter documentation", "billing:read": "Billing overview",
    74→    74→    "billing:write": "Billing administration", "payments:record": "Record payments",
    75→    75→    "payments:void": "Refunds and payment voiding", "inventory:read": "Inventory",
    76→    76→    "inventory:write": "Inventory administration", "pos:sell": "POS / Pharmacy sales",
    77→    77→    "reports:read": "Reports", "reports:export": "Report export and download",
    78→    78→    "ai:ask": "AI assistant", "ai:act": "AI confirmed actions",
    79→    79→    "audit:read": "Audit log", "staff:read": "Staff directory", "staff:manage": "Staff administration",
    80→    80→    "settings:manage": "Facility settings", "roles:manage": "Role management",
    81→    81→    "workflows:manage": "Workflow automation", "lab:operate": "Lab and radiology",
    82→    82→    "inpatient:manage": "Wards and admissions", "portals:manage": "Patient portals",
    83→    83→    "data:manage": "Data import/export",
    84→    84→  };
    85→    85→  const granted = perms
    86→    86→    .filter((p) => (PERMISSIONS as readonly string[]).includes(p))
    87→    87→    .map((p) => moduleOf[p] ?? p)
    88→    88→    .sort();
    89→    89→  const denied = (Object.keys(moduleOf) as string[])
    90→    90→    .filter((p) => !perms.includes(p))
    91→    91→    .map((p) => moduleOf[p])
    92→    92→    .sort();
    93→    93→  return { role, granted, denied };
    94→    94→}

   133→   133→export async function createStaff(
   134→   134→  ctx: { orgId: string; facilityId: string; userId: string; userName: string },
   135→   135→  input: z.infer<typeof createStaffSchema>
   136→   136→) {
   137→   137→  const email = input.email.toLowerCase();
   138→   138→  await resolveStaffRole(ctx.orgId, input.role);
   139→   139→
   140→   140→  const existingStaff = await db.staffMember.findFirst({ where: { orgId: ctx.orgId, facilityId: ctx.facilityId, email } });
   141→   141→  if (existingStaff) throw new AppError(409, "staff_exists", "A staff member with this email already exists at this facility.");
   142→   142→
   143→   143→  const name = displayName(input.firstName, input.lastName);
   144→   144→  let tempPasswordIssued: string | null = null;
   145→   145→  let invitationToken: string | null = null;
   146→   146→
   147→   147→  const staff = await db.$transaction(async (tx) => {
   148→   148→    let userId: string | null = null;
   149→   149→    const existingUser = await tx.user.findUnique({ where: { email } });
   150→   150→
   151→   151→    if (input.accessMode === "TEMP_PASSWORD") {
   152→   152→      if (input.tempPassword) {
   153→   153→        const policy = passwordPolicyError(input.tempPassword);
   154→   154→        if (policy) throw new AppError(400, "weak_password", policy);
   155→   155→      }
   156→   156→      const temp = input.tempPassword ?? generateTempPassword();
   157→   157→      if (existingUser) {
   158→   158→        // Account already exists (cross-facility staff): link it, no password reset
   159→   159→        // through this path - use the password reset flow for account recovery.
   160→   160→        userId = existingUser.id;
   161→   161→      } else {
   162→   162→        const user = await tx.user.create({
   163→   163→          data: { email, name, passwordHash: hashPassword(temp), mustChangePassword: true },
   164→   164→        });
   165→   165→        userId = user.id;
   166→   166→        tempPasswordIssued = temp; // surfaced ONCE to the owner for handover; never logged or stored in plaintext
   167→   167→      }
   168→   168→      const dup = await tx.membership.findFirst({ where: { userId: userId!, orgId: ctx.orgId } });
   169→   169→      if (!dup) {
   170→   170→        await tx.membership.create({ data: { userId: userId!, orgId: ctx.orgId, role: input.role } });
   171→   171→      }
   172→   172→    }
   173→   173→
   174→   174→    const staff = await tx.staffMember.create({
   175→   175→      data: {
   176→   176→        orgId: ctx.orgId, facilityId: ctx.facilityId, userId,
   177→   177→        name, firstName: input.firstName.trim(), lastName: input.lastName.trim(),
   178→   178→        email, phone: input.phone || null, staffNo: input.staffNo || null,
   179→   179→        photoData: input.photoData || null,
   180→   180→        designation: input.designation, role: input.role,
   181→   181→        department: input.department || null, serviceLine: input.serviceLine || null,
   182→   182→        employmentStatus: input.employmentStatus,
   183→   183→        status: input.accessMode === "INVITE" ? "INVITED" : "ACTIVE",
   184→   184→        invitedAt: input.accessMode === "INVITE" ? new Date() : null,
   185→   185→        activatedAt: input.accessMode === "TEMP_PASSWORD" ? new Date() : null,
   186→   186→        active: input.accessMode === "TEMP_PASSWORD",
   187→   187→      },
   188→   188→    });
   189→   189→
   190→   190→    if (input.accessMode === "INVITE") {
   191→   191→      const token = randomBytes(24).toString("hex");
   192→   192→      await tx.passwordResetToken.create({
   193→   193→        data: {
   194→   194→          orgId: ctx.orgId, facilityId: ctx.facilityId, staffId: staff.id, userId: null,
   195→   195→          kind: "INVITE", tokenHash: hashToken(token),
   196→   196→          expiresAt: new Date(Date.now() + 7 * 24 * 3600 * 1000),
   197→   197→          createdById: ctx.userId,
   198→   198→        },
   199→   199→      });
   200→   200→      invitationToken = token;
   201→   201→      await tx.communicationMessage.create({
   202→   202→        data: {
   203→   203→          orgId: ctx.orgId, facilityId: ctx.facilityId, channel: "EMAIL",
   204→   204→          type: "STAFF_INVITATION", recipient: email,
   205→   205→          subject: `You have been invited to join ${name ? "the team" : "Curavilo"}`,
   206→   206→          bodyPreview: `Accept your Curavilo staff invitation and set your own password using this single-use code: ${token}`,
   207→   207→          status: "SIMULATED", provider: "console", source: "MANUAL", sourceId: staff.id,
   208→   208→        },
   209→   209→      }).catch(() => undefined);
   210→   210→    }
   211→   211→
   212→   212→    await tx.auditEvent.create({
   213→   213→      data: {
   214→   214→        orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   215→   215→        action: input.accessMode === "INVITE" ? "staff.invitation_created" : "staff.created",
   216→   216→        entityType: "StaffMember", entityId: staff.id,
   217→   217→        summary: `${name} added as ${input.role} (${input.accessMode === "INVITE" ? "invitation sent" : "temporary password issued"}).`,
   218→   218→      },
   219→   219→    });
   220→   220→
   221→   221→    return staff;
   222→   222→  });
   223→   223→
   224→   224→  return {
   225→   225→    staff,
   226→   226→    tempPassword: tempPasswordIssued,
   227→   227→    invitationToken,
   228→   228→    inviteUrl: invitationToken ? `/staff/accept?token=${invitationToken}` : null,
   229→   229→  };
   230→   230→}

   232→   232→export async function getStaffDetail(orgId: string, facilityId: string, staffId: string) {
   233→   233→  const staff = await db.staffMember.findFirst({ where: { id: staffId, orgId, facilityId } });
   234→   234→  if (!staff) throw NotFound("Staff member not found.");
   235→   235→  const user = staff.userId ? await db.user.findUnique({ where: { id: staff.userId }, include: { overrides: true } }) : null;
   236→   236→  const activity = await db.auditEvent.findMany({
   237→   237→    where: staff.userId ? { orgId, OR: [{ entityId: staff.id, entityType: "StaffMember" }, { actorUserId: staff.userId }] } : { orgId, entityId: staff.id },
   238→   238→    orderBy: { createdAt: "desc" },
   239→   239→    take: 25,
   240→   240→    select: { id: true, action: true, summary: true, createdAt: true, actorLabel: true },
   241→   241→  });
   242→   242→  const recentLogins = user?.lastLoginAt ? [user.lastLoginAt] : [];
   243→   243→  return {
   244→   244→    staff: {
   245→   245→      id: staff.id, userId: staff.userId, name: staff.name, firstName: staff.firstName,
   246→   246→      lastName: staff.lastName, email: staff.email, phone: staff.phone, staffNo: staff.staffNo,
   247→   247→      photoData: staff.photoData, designation: staff.designation, role: staff.role,
   248→   248→      department: staff.department, serviceLine: staff.serviceLine,
   249→   249→      employmentStatus: staff.employmentStatus, status: staff.status, active: staff.active,
   250→   250→      invitedAt: staff.invitedAt, activatedAt: staff.activatedAt, deactivatedAt: staff.deactivatedAt,
   251→   251→      createdAt: staff.createdAt,
   252→   252→    },
   253→   253→    account: user ? {
   254→   254→      email: user.email, lastLoginAt: user.lastLoginAt, mustChangePassword: user.mustChangePassword,
   255→   255→      lockedUntil: user.lockedUntil,
   256→   256→      overrides: user.overrides.filter((o) => o.orgId === orgId).map((o) => ({ permission: o.permission, effect: o.effect })),
   257→   257→    } : null,
   258→   258→    permissions: staff.role ? permissionsForRole(staff.role) : [],
   259→   259→    activity,
   260→   260→    recentLogins,
   261→   261→  };
   262→   262→}

   264→   264→export async function updateStaff(
   265→   265→  ctx: { orgId: string; facilityId: string; userId: string; userName: string },
   266→   266→  staffId: string,
   267→   267→  input: z.infer<typeof updateStaffSchema>
   268→   268→) {
   269→   269→  const staff = await db.staffMember.findFirst({ where: { id: staffId, orgId: ctx.orgId, facilityId: ctx.facilityId } });
   270→   270→  if (!staff) throw NotFound("Staff member not found.");
   271→   271→  if (input.role) await resolveStaffRole(ctx.orgId, input.role);
   272→   272→
   273→   273→  const data: Record<string, unknown> = {};
   274→   274→  if (input.firstName || input.lastName) {
   275→   275→    const first = input.firstName ?? staff.firstName ?? "";
   276→   276→    const last = input.lastName ?? staff.lastName ?? "";
   277→   277→    data.firstName = first.trim();
   278→   278→    data.lastName = last.trim();
   279→   279→    data.name = displayName(first || "Staff", last || "Member");
   280→   280→  }
   281→   281→  for (const k of ["phone", "staffNo", "photoData", "designation", "department", "serviceLine", "employmentStatus"] as const) {
   282→   282→    if (input[k] !== undefined) data[k] = input[k] || null;
   283→   283→  }
   284→   284→  const roleChanged = !!(input.role && input.role !== staff.role);
   285→   285→  if (input.role) data.role = input.role;
   286→   286→
   287→   287→  const updated = await db.$transaction(async (tx) => {
   288→   288→    const s = await tx.staffMember.update({ where: { id: staff.id }, data });
   289→   289→    if (roleChanged && s.userId) {
   290→   290→      await tx.membership.updateMany({
   291→   291→        where: { userId: s.userId, orgId: ctx.orgId },
   292→   292→        data: { role: input.role! },
   293→   293→      });
   294→   294→      // Role changes revoke existing sessions so new permissions take effect cleanly.
   295→   295→      await tx.session.deleteMany({ where: { userId: s.userId } });
   296→   296→    }
   297→   297→    await tx.auditEvent.create({
   298→   298→      data: {
   299→   299→        orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   300→   300→        action: roleChanged ? "staff.role_changed" : "staff.updated",
   301→   301→        entityType: "StaffMember", entityId: staff.id,
   302→   302→        summary: roleChanged
   303→   303→          ? `${staff.name} role changed from ${staff.role ?? "?"} to ${input.role}.`
   304→   304→          : `${staff.name} profile updated.`,
   305→   305→      },
   306→   306→    });
   307→   307→    return s;
   308→   308→  });
   309→   309→  return updated;
   310→   310→}

   312→   312→async function transitionStatus(
   313→   313→  ctx: { orgId: string; facilityId: string; userId: string; userName: string },
   314→   314→  staffId: string,
   315→   315→  newStatus: string,
   316→   316→  summaryDetail: string,
   317→   317→  opts: { revokeSessions?: boolean; setActive?: boolean } = {}
   318→   318→) {
   319→   319→  const staff = await db.staffMember.findFirst({ where: { id: staffId, orgId: ctx.orgId, facilityId: ctx.facilityId } });
   320→   320→  if (!staff) throw NotFound("Staff member not found.");
   321→   321→  if (!STAFF_STATUSES.includes(newStatus as never)) throw BadRequest("Invalid status.");
   322→   322→  const data: Record<string, unknown> = { status: newStatus };
   323→   323→  if (newStatus === "ACTIVE") { data.activatedAt = new Date(); data.active = true; }
   324→   324→  if (["INACTIVE", "SUSPENDED", "TERMINATED"].includes(newStatus)) { data.deactivatedAt = new Date(); data.active = false; }
   325→   325→  if (newStatus === "LOCKED") data.active = false;
   326→   326→
   327→   327→  await db.$transaction(async (tx) => {
   328→   328→    await tx.staffMember.update({ where: { id: staff.id }, data });
   329→   329→    if (opts.revokeSessions && staff.userId) {
   330→   330→      // Deactivated personnel must not ride an old session (spec §57, §76):
   331→   331→      // DB-backed sessions are deleted server-side, killing every device.
   332→   332→      await tx.session.deleteMany({ where: { userId: staff.userId } });
   333→   333→    }
   334→   334→    await tx.auditEvent.create({
   335→   335→      data: {
   336→   336→        orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   337→   337→        action: "staff.status_changed", entityType: "StaffMember", entityId: staff.id,
   338→   338→        summary: `${staff.name} -> ${newStatus}${summaryDetail ? ` (${summaryDetail})` : ""}.`,
   339→   339→        metadata: opts.revokeSessions ? JSON.stringify({ sessionsRevoked: true }) : null,
   340→   340→      },
   341→   341→    });
   342→   342→  });
   343→   343→  return db.staffMember.findUnique({ where: { id: staff.id } });
   344→   344→}

   346→   346→export async function staffAction(
   347→   347→  ctx: { orgId: string; facilityId: string; userId: string; userName: string },
   348→   348→  staffId: string,
   349→   349→  input: z.infer<typeof staffActionSchema>
   350→   350→) {
   351→   351→  const staff = await db.staffMember.findFirst({ where: { id: staffId, orgId: ctx.orgId, facilityId: ctx.facilityId } });
   352→   352→  if (!staff) throw NotFound("Staff member not found.");
   353→   353→
   354→   354→  switch (input.action) {
   355→   355→    case "deactivate":
   356→   356→      return transitionStatus(ctx, staffId, "INACTIVE", input.reason ?? "", { revokeSessions: true });
   357→   357→    case "reactivate":
   358→   358→      return transitionStatus(ctx, staffId, "ACTIVE", input.reason ?? "");
   359→   359→    case "suspend":
   360→   360→      return transitionStatus(ctx, staffId, "SUSPENDED", input.reason ?? "", { revokeSessions: true });
   361→   361→    case "reinstate":
   362→   362→      return transitionStatus(ctx, staffId, "ACTIVE", input.reason ?? "");
   363→   363→    case "terminate":
   364→   364→      return transitionStatus(ctx, staffId, "TERMINATED", input.reason ?? "", { revokeSessions: true });
   365→   365→    case "lock":
   366→   366→      return transitionStatus(ctx, staffId, "LOCKED", input.reason ?? "", { revokeSessions: true });
   367→   367→    case "unlock": {
   368→   368→      if (staff.userId) {
   369→   369→        await db.user.update({ where: { id: staff.userId }, data: { failedCount: 0, lockedUntil: null } });
   370→   370→      }
   371→   371→      await db.staffMember.update({ where: { id: staff.id }, data: { status: "ACTIVE", active: true } });
   372→   372→      await audit({
   373→   373→        orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   374→   374→        action: "staff.status_changed", entityType: "StaffMember", entityId: staff.id,
   375→   375→        summary: `${staff.name} unlocked and reinstated to ACTIVE.`,
   376→   376→      });
   377→   377→      return db.staffMember.findUnique({ where: { id: staff.id } });
   378→   378→    }
   379→   379→    case "issue_temp_password": {
   380→   380→      if (!staff.userId) throw BadRequest("This staff member has no account yet; send an invitation first.");
   381→   381→      const temp = input.tempPassword ?? generateTempPassword();
   382→   382→      const policy = passwordPolicyError(temp);
   383→   383→      if (policy) throw new AppError(400, "weak_password", policy);
   384→   384→      await db.$transaction([
   385→   385→        db.user.update({
   386→   386→          where: { id: staff.userId },
   387→   387→          data: { passwordHash: hashPassword(temp), mustChangePassword: true, failedCount: 0, lockedUntil: null },
   388→   388→        }),
   389→   389→        db.session.deleteMany({ where: { userId: staff.userId } }),
   390→   390→        db.auditEvent.create({
   391→   391→          data: {
   392→   392→            orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   393→   393→            action: "staff.temp_password_issued", entityType: "StaffMember", entityId: staff.id,
   394→   394→            summary: `Temporary password issued for ${staff.name}; change forced at next login.`,
   395→   395→          },
   396→   396→        }),
   397→   397→      ]);
   398→   398→      return { ok: true, tempPassword: temp };
   399→   399→    }
   400→   400→    case "send_invitation": {
   401→   401→      const token = randomBytes(24).toString("hex");
   402→   402→      await db.$transaction(async (tx) => {
   403→   403→        await tx.passwordResetToken.create({
   404→   404→          data: {
   405→   405→            orgId: ctx.orgId, facilityId: ctx.facilityId, staffId: staff.id, userId: null,
   406→   406→            kind: "INVITE", tokenHash: hashToken(token),
   407→   407→            expiresAt: new Date(Date.now() + 7 * 24 * 3600 * 1000),
   408→   408→            createdById: ctx.userId,
   409→   409→          },
   410→   410→        });
   411→   411→        await tx.staffMember.update({ where: { id: staff.id }, data: { status: "INVITED", invitedAt: new Date() } });
   412→   412→        await tx.communicationMessage.create({
   413→   413→          data: {
   414→   414→            orgId: ctx.orgId, facilityId: ctx.facilityId, channel: "EMAIL",
   415→   415→            type: "STAFF_INVITATION", recipient: staff.email ?? "",
   416→   416→            subject: "Your Curavilo staff invitation",
   417→   417→            bodyPreview: `Set your Curavilo password with this single-use invitation code: ${token}`,
   418→   418→            status: "SIMULATED", provider: "console", source: "MANUAL", sourceId: staff.id,
   419→   419→          },
   420→   420→        }).catch(() => undefined);
   421→   421→        await tx.auditEvent.create({
   422→   422→          data: {
   423→   423→            orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   424→   424→            action: "staff.invitation_created", entityType: "StaffMember", entityId: staff.id,
   425→   425→            summary: `Invitation re-sent to ${staff.email}.`,
   426→   426→          },
   427→   427→        });
   428→   428→      });
   429→   429→      return { ok: true, invitationToken: token, inviteUrl: `/staff/accept?token=${token}` };
   430→   430→    }
   431→   431→    case "set_override": {
   432→   432→      if (!input.permission || !input.effect) throw BadRequest("permission and effect are required.");
   433→   433→      if (!(PERMISSIONS as readonly string[]).includes(input.permission)) throw BadRequest("Unknown permission.");
   434→   434→      if (PROTECTED_PERMISSIONS.includes(input.permission as never)) {
   435→   435→        throw Forbidden("This permission is platform-protected and cannot be overridden.");
   436→   436→      }
   437→   437→      if (!staff.userId) throw BadRequest("This staff member has no account yet.");
   438→   438→      await db.permissionOverride.upsert({
   439→   439→        where: { userId_orgId_permission: { userId: staff.userId, orgId: ctx.orgId, permission: input.permission } },
   440→   440→        create: { userId: staff.userId, orgId: ctx.orgId, permission: input.permission, effect: input.effect },
   441→   441→        update: { effect: input.effect },
   442→   442→      });
   443→   443→      if (staff.userId) await db.session.deleteMany({ where: { userId: staff.userId } }); // re-resolve on next login
   444→   444→      await audit({
   445→   445→        orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   446→   446→        action: "staff.permission_changed", entityType: "StaffMember", entityId: staff.id,
   447→   447→        summary: `Permission override ${input.effect} ${input.permission} for ${staff.name}.`,
   448→   448→      });
   449→   449→      return { ok: true };
   450→   450→    }
   451→   451→    case "clear_override": {
   452→   452→      if (!input.permission) throw BadRequest("permission is required.");
   453→   453→      if (!staff.userId) throw BadRequest("This staff member has no account yet.");
   454→   454→      await db.permissionOverride.deleteMany({
   455→   455→        where: { userId: staff.userId, orgId: ctx.orgId, permission: input.permission },
   456→   456→      });
   457→   457→      await db.session.deleteMany({ where: { userId: staff.userId } });
   458→   458→      await audit({
   459→   459→        orgId: ctx.orgId, facilityId: ctx.facilityId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   460→   460→        action: "staff.permission_changed", entityType: "StaffMember", entityId: staff.id,
   461→   461→        summary: `Permission override cleared (${input.permission}) for ${staff.name}.`,
   462→   462→      });
   463→   463→      return { ok: true };
   464→   464→    }
   465→   465→  }
   466→   466→}

   470→   470→export const customRoleSchema = z.object({
   471→   471→  name: z.string().min(2).max(60),
   472→   472→  description: z.string().max(200).optional().nullable(),
   473→   473→  permissions: z.array(z.string()).max(PERMISSIONS.length),

   475→   475→
   476→   476→export function validateCustomRolePermissions(permissions: string[]): string[] {
   477→   477→  const valid: string[] = [];
   478→   478→  for (const p of permissions) {
   479→   479→    if (!(PERMISSIONS as readonly string[]).includes(p)) continue;
   480→   480→    if (PROTECTED_PERMISSIONS.includes(p as never)) continue; // silently drop protected
   481→   481→    if (!valid.includes(p)) valid.push(p);
   482→   482→  }
   483→   483→  return valid;
   484→   484→}

   486→   486→export async function listRoles(orgId: string) {
   487→   487→  const customs = await db.customRole.findMany({ where: { orgId }, orderBy: { name: "asc" } });
   488→   488→  const builtIn = ROLES.map((r) => ({
   489→   489→    name: r, builtIn: true as const, description: null,
   490→   490→    permissions: permissionsForRole(r), systemProtected: r === "OWNER",
   491→   491→  }));
   492→   492→  const custom = customs.map((c) => ({
   493→   493→    name: c.name, builtIn: false as const, description: c.description,
   494→   494→    permissions: (() => { try { return JSON.parse(c.permissions) as string[]; } catch { return []; } })(),
   495→   495→    systemProtected: false,
   496→   496→  }));
   497→   497→  return { builtIn, custom, allPermissions: PERMISSIONS, protectedPermissions: PROTECTED_PERMISSIONS };
   498→   498→}

   500→   500→export async function createCustomRole(
   501→   501→  ctx: { orgId: string; userId: string; userName: string },
   502→   502→  input: z.infer<typeof customRoleSchema>
   503→   503→) {
   504→   504→  const name = input.name.trim();
   505→   505→  if ((ROLES as readonly string[]).includes(name)) throw BadRequest("This name is reserved for built-in roles.");
   506→   506→  const dup = await db.customRole.findFirst({ where: { orgId: ctx.orgId, name } });
   507→   507→  if (dup) throw new AppError(409, "role_exists", "A role with this name already exists.");
   508→   508→  const permissions = validateCustomRolePermissions(input.permissions);
   509→   509→  const role = await db.customRole.create({
   510→   510→    data: {
   511→   511→      orgId: ctx.orgId, name,
   512→   512→      description: input.description || null,
   513→   513→      permissions: JSON.stringify(permissions),
   514→   514→      createdById: ctx.userId,
   515→   515→    },
   516→   516→  });
   517→   517→  await audit({
   518→   518→    orgId: ctx.orgId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   519→   519→    action: "roles.created", entityType: "CustomRole", entityId: role.id,
   520→   520→    summary: `Custom role "${name}" created with ${permissions.length} permissions.`,
   521→   521→  });
   522→   522→  return role;
   523→   523→}

   525→   525→export async function updateCustomRole(
   526→   526→  ctx: { orgId: string; userId: string; userName: string },
   527→   527→  name: string,
   528→   528→  input: z.infer<typeof customRoleSchema>
   529→   529→) {
   530→   530→  const existing = await db.customRole.findFirst({ where: { orgId: ctx.orgId, name } });
   531→   531→  if (!existing) throw NotFound("Custom role not found.");
   532→   532→  const permissions = validateCustomRolePermissions(input.permissions);
   533→   533→  const updated = await db.customRole.update({
   534→   534→    where: { id: existing.id },
   535→   535→    data: {
   536→   536→      description: input.description || null,
   537→   537→      permissions: JSON.stringify(permissions),
   538→   538→    },
   539→   539→  });
   540→   540→  // Force re-resolution: users of this role get updated permissions on next login.

   542→   542→  for (const u of users) await db.session.deleteMany({ where: { userId: u.userId } });
   543→   543→  await audit({
   544→   544→    orgId: ctx.orgId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   545→   545→    action: "roles.updated", entityType: "CustomRole", entityId: existing.id,
   546→   546→    summary: `Custom role "${name}" updated (${permissions.length} permissions; sessions re-resolved).`,
   547→   547→  });
   548→   548→  return updated;
   549→   549→}

   552→   552→  ctx: { orgId: string; userId: string; userName: string },
   553→   553→  name: string
   554→   554→) {
   555→   555→  const existing = await db.customRole.findFirst({ where: { orgId: ctx.orgId, name } });

   561→   561→  }
   562→   562→  await db.customRole.delete({ where: { id: existing.id } });
   563→   563→  await audit({
   564→   564→    orgId: ctx.orgId, actorUserId: ctx.userId, actorLabel: ctx.userName,
   565→   565→    action: "roles.deleted", entityType: "CustomRole", entityId: existing.id,
   566→   566→    summary: `Custom role "${name}" deleted.`,
   567→   567→  });
   568→   568→  return { ok: true };
